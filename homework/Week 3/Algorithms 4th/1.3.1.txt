1. Phân tích:
- Trong lớp FixedCapacityStackOfStrings, mảng `a[]` lưu trữ các phần tử có dung lượng tối đa cố định (capacity) là `a.length`.
- Biến `N` đại diện cho số lượng phần tử hiện có trong stack.
- Stack đầy (isFull) khi số phần tử hiện tại `N` bằng với kích thước tối đa của mảng `a.length`.

2. Phương thức isFull():
   public boolean isFull() {
       return N == a.length;
   }

3. Lớp FixedCapacityStackOfStrings hoàn chỉnh:

public class FixedCapacityStackOfStrings {
    private String[] a; // Mảng chứa các phần tử
    private int N;      // Số lượng phần tử hiện tại trong stack

    // Khởi tạo stack với dung lượng cố định cap
    public FixedCapacityStackOfStrings(int cap) {
        a = new String[cap];
        N = 0;
    }

    // Kiểm tra stack có rỗng hay không
    public boolean isEmpty() {
        return N == 0;
    }

    // Kiểm tra stack đã đầy chưa (Yêu cầu bài 1.3.1)
    public boolean isFull() {
        return N == a.length;
    }

    // Trả về số lượng phần tử hiện tại
    public int size() {
        return N;
    }

    // Thêm phần tử vào stack
    public void push(String item) {
        if (isFull()) {
            throw new RuntimeException("Stack is full");
        }
        a[N++] = item;
    }

    // Lấy phần tử ra khỏi stack
    public String pop() {
        if (isEmpty()) {
            throw new RuntimeException("Stack is empty");
        }
        String item = a[--N];
        a[N] = null; // Tránh loitering (giữ lại tham chiếu không cần thiết)
        return item;
    }

    public static void main(String[] args) {
        FixedCapacityStackOfStrings stack = new FixedCapacityStackOfStrings(3);

        System.out.println("Stack empty? " + stack.isEmpty()); // true
        System.out.println("Stack full? " + stack.isFull());   // false

        stack.push("Hello");
        stack.push("World");
        stack.push("Java");

        System.out.println("Stack full sau khi push 3 phan tu? " + stack.isFull()); // true

        System.out.println("Pop: " + stack.pop());
        System.out.println("Stack full sau khi pop 1 phan tu? " + stack.isFull());   // false
    }
}
